find_package(Threads REQUIRED)
add_library(opennova_update_core STATIC client.cpp options.cpp)
target_include_directories(opennova_update_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(opennova_update_core PUBLIC opennova_app_common Threads::Threads)
target_compile_features(opennova_update_core PUBLIC cxx_std_17)

add_executable(opennova_update main.cpp platform.cpp)
set_target_properties(opennova_update PROPERTIES OUTPUT_NAME update)
target_link_libraries(opennova_update PRIVATE opennova_update_core)
if(WIN32)
    set_target_properties(opennova_update PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(opennova_update PRIVATE comctl32 shell32)
endif()
if(MSVC)
    target_sources(opennova_update PRIVATE ${CMAKE_SOURCE_DIR}/apps/3di/utf8.manifest)
endif()
